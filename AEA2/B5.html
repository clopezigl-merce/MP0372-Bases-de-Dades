A Oracle no funcionen NOW() ni DATEADD(): s’usa SYSDATE i la taula dual.
SELECT SYSDATE FROM dual;
SELECT SYSDATE + INTERVAL '10' DAY FROM dual;
SELECT UPPER(last_name), LENGTH(last_name) FROM employees;
SELECT MONTHS_BETWEEN(SYSDATE, hire_date) FROM employees;
