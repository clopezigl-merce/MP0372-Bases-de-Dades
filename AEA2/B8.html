B8 · Funcions de grup

GROUP BY, HAVING i agregats.

← AEA2

Ordre bàsic d’una SELECT

SELECT camp, MAX(camp2)
FROM taula
WHERE condició      -- actua sobre cada fila
GROUP BY camps
HAVING condició     -- actua sobre el grup
ORDER BY camp;
Amb GROUP BY cal seleccionar el camp d’agrupació. WHERE filtra files; HAVING filtra grups.

Exemples

SELECT job_id, COUNT(*) "Núm. d'empleats"
FROM employees
GROUP BY job_id
ORDER BY 2;

SELECT job_id, COUNT(*)
FROM employees
WHERE last_name LIKE '%&PideLetra%'
GROUP BY job_id
HAVING COUNT(*) > 2
ORDER BY 2;
